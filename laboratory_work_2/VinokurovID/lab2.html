MEAN = (X_train.float() / 255).mean(dim=(0, 2, 3)).view(1, 3, 1, 1)
STD = (X_train.float() / 255).std(dim=(0, 2, 3)).view(1, 3, 1, 1)


def normalize(x_uint8):
    return (x_uint8.float() / 255 - MEAN) / STD


def augment(x, pad=4, cut=16):
    n, c, h, w = x.shape
    dev = x.device
    x = F.pad(x, (pad, pad, pad, pad))
    dy = torch.randint(0, 2 * pad + 1, (n,), device=dev)
    dx = torch.randint(0, 2 * pad + 1, (n,), device=dev)
    rows = (dy[:, None] + torch.arange(h, device=dev))[:, None, :, None]
    cols = (dx[:, None] + torch.arange(w, device=dev))[:, None, None, :]
    x = x[torch.arange(n, device=dev)[:, None, None, None], torch.arange(c, device=dev)[None, :, None, None], rows, cols]
    flip = torch.rand(n, device=dev) < 0.5
    x = torch.where(flip[:, None, None, None], x.flip(3), x)
    cy = torch.randint(0, h, (n, 1, 1), device=dev)
    cx = torch.randint(0, w, (n, 1, 1), device=dev)
    ys = torch.arange(h, device=dev)[None, :, None]
    xs = torch.arange(w, device=dev)[None, None, :]
    hole = ((ys - cy).abs() < cut // 2) & ((xs - cx).abs() < cut // 2)
    return x * (~hole)[:, None].to(x.dtype)

class ResidualBlock(nn.Module):

    def __init__(self, in_channels, out_channels, stride=1):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3, stride, 1, bias=False)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3, 1, 1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels)
        self.shortcut = nn.Identity()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, 1, stride, bias=False),
                nn.BatchNorm2d(out_channels),
            )

    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        return F.relu(out + self.shortcut(x))


class CifarResNet(nn.Module):

    def __init__(self, num_classes=10, widths=(64, 128, 256, 512), depths=(2, 2, 2, 2)):
        super().__init__()
        self.stem = nn.Sequential(
            nn.Conv2d(3, widths[0], 3, 1, 1, bias=False),
            nn.BatchNorm2d(widths[0]),
            nn.ReLU(inplace=True),
        )
        blocks = []
        in_channels = widths[0]
        for stage, (width, depth) in enumerate(zip(widths, depths)):
            for k in range(depth):
                stride = 2 if stage > 0 and k == 0 else 1
                blocks.append(ResidualBlock(in_channels, width, stride))
                in_channels = width
        self.blocks = nn.Sequential(*blocks)
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Linear(in_channels, num_classes)

    def forward(self, x):
        x = self.blocks(self.stem(x))
        return self.fc(torch.flatten(self.pool(x), 1))

EPOCHS = 100
BATCH_SIZE = 128
LEARNING_RATE = 0.1
MOMENTUM = 0.9
WEIGHT_DECAY = 5e-4
LABEL_SMOOTHING = 0.1

use_amp = device.type == "cuda"


def evaluate(model, X, y, batch_size=1000):
    model.eval()
    total_loss, correct = 0.0, 0
    with torch.no_grad(), torch.autocast(device_type=device.type, dtype=torch.float16, enabled=use_amp):
        for start in range(0, X.shape[0], batch_size):
            xb = normalize(X[start:start + batch_size]).contiguous(memory_format=torch.channels_last)
            yb = y[start:start + batch_size]
            logits = model(xb)
            total_loss += F.cross_entropy(logits.float(), yb, reduction="sum").item()
            correct += (logits.argmax(dim=1) == yb).sum().item()
    acc = correct / X.shape[0]
    return total_loss / X.shape[0], acc, 1.0 - acc


def train(model, X, y, epochs, batch_size, lr):
    optimizer = torch.optim.SGD(model.parameters(), lr=lr, momentum=MOMENTUM,
                                weight_decay=WEIGHT_DECAY, nesterov=True)
    steps_per_epoch = (X.shape[0] + batch_size - 1) // batch_size
    scheduler = torch.optim.lr_scheduler.OneCycleLR(optimizer, max_lr=lr, epochs=epochs,
                                                    steps_per_epoch=steps_per_epoch, pct_start=0.15)
    scaler = torch.amp.GradScaler(device.type, enabled=use_amp)
    criterion = nn.CrossEntropyLoss(label_smoothing=LABEL_SMOOTHING)
    history = {"train_loss": [], "train_error": [], "test_error": [], "time": []}

    for epoch in range(1, epochs + 1):
        model.train()
        t0 = time.perf_counter()
        perm = torch.randperm(X.shape[0], device=X.device)
        for start in range(0, X.shape[0], batch_size):
            idx = perm[start:start + batch_size]
            xb = augment(normalize(X[idx])).contiguous(memory_format=torch.channels_last)
            with torch.autocast(device_type=device.type, dtype=torch.float16, enabled=use_amp):
                loss = criterion(model(xb), y[idx])
            optimizer.zero_grad(set_to_none=True)
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            scheduler.step()
        if device.type == "cuda":
            torch.cuda.synchronize()
        epoch_time = time.perf_counter() - t0

        train_loss, train_acc, train_err = evaluate(model, X, y)
        _, test_acc, test_err = evaluate(model, X_test, y_test)
        history["train_loss"].append(train_loss)
        history["train_error"].append(train_err)
        history["test_error"].append(test_err)
        history["time"].append(epoch_time)
        print(f"Эпоха {epoch:3d}/{epochs} | train: loss = {train_loss:.4f}, ошибка = {train_err * 100:5.2f}%, "
              f"точность = {train_acc * 100:5.2f}% | test: точность = {test_acc * 100:5.2f}% | "
              f"lr = {scheduler.get_last_lr()[0]:.4f} | время = {epoch_time:.1f} с")
    return history
