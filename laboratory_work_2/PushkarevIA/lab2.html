# Функция для денормализации (чтобы картинки выглядели естественно)
def denormalize(img):
    img = img.numpy().transpose((1, 2, 0))
    img = np.array(CIFAR_STD) * img + np.array(CIFAR_MEAN)
    return np.clip(img, 0, 1)

class VGGLikeCNN(nn.Module):
    """3 сверточных блока (64→128→256), каждый: 2×(Conv3×3–BN–ReLU) + MaxPool.
       Затем Global Average Pooling, Dropout и линейный слой на 10 классов."""
    def __init__(self, num_classes=10):
        super().__init__()
        # Блок 1: 3 -> 64
        self.block1 = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.Conv2d(64, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2)  # 32 -> 16
        )
        # Блок 2: 64 -> 128
        self.block2 = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),
            nn.Conv2d(128, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2)  # 16 -> 8
        )
        # Блок 3: 128 -> 256
        self.block3 = nn.Sequential(
            nn.Conv2d(128, 256, kernel_size=3, padding=1),
            nn.BatchNorm2d(256),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, kernel_size=3, padding=1),
            nn.BatchNorm2d(256),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2)  # 8 -> 4
        )
        # GAP + Dropout + Linear
        self.gap = nn.AdaptiveAvgPool2d(1)
        self.dropout = nn.Dropout(0.3)
        self.fc = nn.Linear(256, num_classes)

    def forward(self, x):
        x = self.block1(x)
        x = self.block2(x)
        x = self.block3(x)
        x = self.gap(x)          # (B, 256, 1, 1)
        x = torch.flatten(x, 1)  # (B, 256)
        x = self.dropout(x)
        x = self.fc(x)
        return x

def calculate_accuracy(output, target):
    preds = output.argmax(dim=1)
    return (preds == target).sum().item()

def evaluate(model, loader):
    """Возвращает (loss, accuracy) на переданном загрузчике."""
    model.eval()
    total_loss = 0.0
    correct = 0
    total = 0
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)
            total_loss += loss.item() * images.size(0)
            correct += calculate_accuracy(outputs, labels)
            total += images.size(0)
    avg_loss = total_loss / total
    acc = correct / total
    return avg_loss, acc

EPOCHS = 20
train_losses = []
train_accs = []
test_losses = []
test_accs = []

print(f"{'Эпоха':>5} | {'Train Loss':>10} | {'Train Acc':>9} | {'Test Loss':>9} | {'Test Acc':>8} | {'Время (с)':>9}")
print("-" * 70)

for epoch in range(1, EPOCHS + 1):
    start_time = time.time()
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0

    for images, labels in train_loader:
        images, labels = images.to(device), labels.to(device)

        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        scheduler.step()  # OneCycleLR обновляется каждый шаг

        running_loss += loss.item() * images.size(0)
        correct += calculate_accuracy(outputs, labels)
        total += images.size(0)

    train_loss = running_loss / total
    train_acc = correct / total
    train_losses.append(train_loss)
    train_accs.append(train_acc)

    # Оценка на тесте после каждой эпохи
    test_loss, test_acc = evaluate(model, test_loader)
    test_losses.append(test_loss)
    test_accs.append(test_acc)

    elapsed = time.time() - start_time
    print(f"{epoch:5d} | {train_loss:10.4f} | {train_acc*100:8.2f}% | "
          f"{test_loss:9.4f} | {test_acc*100:7.2f}% | {elapsed:9.2f}")

# Собираем предсказания по всему тестовому набору и находим ошибки
model.eval()
all_images = []
all_preds = []
all_labels = []

with torch.no_grad():
    for images, labels in test_loader:
        images_dev = images.to(device)
        outputs = model(images_dev)
        preds = outputs.argmax(dim=1).cpu()
        all_images.append(images.cpu())
        all_preds.append(preds)
        all_labels.append(labels)

all_images = torch.cat(all_images)
all_preds  = torch.cat(all_preds)
all_labels = torch.cat(all_labels)
